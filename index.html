    function highlightCode(code, lang) {

      const esc = escapeHtml(code);

      if (lang === 'html') {

        const htmlRx = /(&lt;!--[\s\S]*?--&gt;)|(&lt;\/?[a-zA-Z0-9-]+)|(\s[a-zA-Z0-9-]+)=|(&quot;(?:\\.|(?!&quot;).)*?&quot;)/g;

        return esc.replace(htmlRx, (m, cm, tag, attr, str) => {

          if (cm) return `<span class="hl-cm">${cm}</span>`;

          if (tag) return `<span class="hl-kw">${tag}</span>`;

          if (attr) return `<span class="hl-attr">${attr}</span>=`;

          if (str) return `<span class="hl-str">${str}</span>`;

          return m;

        });

      }

      const rx = lang === 'python'

        ? /(#[^\n]*)|(&quot;(?:\\.|(?!&quot;).)*?&quot;|'(?:\\.|[^'])*')|(@[a-zA-Z_]\w*)|\b(async|await|case|class|def|else|False|for|from|if|import|in|is|lazy|match|None|return|True|while|with|yield)\b|\b(frozendict|sentinel|Path|dict|list|str|int|float|bool|print)\b|\b(\d+(?:\.\d+)?)\b/g

        : /(\/\/[^\n]*)|(&quot;(?:\\.|(?!&quot;).)*?&quot;|'(?:\\.|[^'])*')|(#\s*(?:include|define|ifdef|endif)\b[^\n]*)|\b(auto|async|await|bool|case|class|const|constexpr|double|else|enum|false|field|for|if|int|is|lock|new|null|nullptr|public|private|readonly|record|return|static|string|struct|switch|true|using|var|void|while|with)\b|\b(std::\w+|List|Span|ReadOnlySpan|Task|Lock|Console|HttpClient|CancellationToken|uint8_t)\b|\b(\d+(?:\.\d+)?)\b/g;

      return esc.replace(rx, (m, cm, str, attr, kw, type, num) => {

        if (cm) return `<span class="hl-cm">${cm}</span>`;

        if (str) return `<span class="hl-str">${str}</span>`;

    function shuffle(arr) {

      const a = [...arr];

      for (let i = a.length - 1; i > 0; i--) {

        const j = Math.floor(Math.random() * (i + 1));

        [a[i], a[j]] = [a[j], a[i]];

      }

      return a;

    }

    function handleQuizAnswer(selIdx) {

      if (ok) {

        state.quizScore++;

        state.wrongQuizIds.delete(item.id);

      } else {

        state.wrongQuizIds.add(item.id);

      }

      persistState();

      updateQuizModeLabels();

      document.getElementById('quizScore').textContent = `Счёт: ${state.quizScore}`;

      document.querySelectorAll('.quiz-opt-btn').forEach(b => {

        if (i === item.answer) b.classList.add('border-emerald-600', 'bg-emerald-100/80', 'dark:bg-emerald-500/25', 'font-medium');

        else if (i === selIdx && !ok) b.classList.add('border-rose-600', 'bg-rose-100/80', 'dark:bg-rose-500/25');

        else b.classList.add('opacity-50');

      });

      const fb = document.getElementById('quizFeedback');

    }
